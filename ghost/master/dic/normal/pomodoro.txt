//******************************************************************************
// ポモドーロ（集中タイマー）辞書
//
// 実用：集中 → 休憩 の鐘は、いつも正確に鳴る。
// お遊び：集中が終わったとき、低確率でぽにゃが休憩に入れず、没頭したまま終わらない。
//
// 状態変数（ゴースト終了時に保存される）
//   pomo_state       0=待機 / 1=集中中 / 2=休憩中
//   pomo_end         今の区間が終わる時刻（GETSECCOUNT の絶対時刻）
//   pomo_mid         延長中の実況を出す時刻（0 なら無し）
//   pomo_talk_next   次の集中専用トークを出す時刻（0 なら出さない）
//   pomo_focus_min   集中の分数（定番は 25）
//   pomo_break_min   休憩の分数（定番は 5）
//   pomo_overrun     1 ならこの休憩はぽにゃが終わらない回
//   pomo_type        延長の型（0=あと一行だけ / 1=タイマー改良に吸われる）
//   pomo_oneline     型 0 で「あと一行」を言った回数（実況と止めで使う）
//   pomo_miss        延長が起きなかった回数（確率の補正に使う）
//   pomo_aitalk_backup / pomo_interval_override
//                    集中中は aitalkinterval を 0 にして本体のランダムトークを止める。
//                    元の値を退避しておき、集中が終わったら戻す。
//   pomo_ring_shown  シェルに出している進捗リングの着せ替え（'カテゴリ,パーツ'。無ければ ''）
//
// 進捗リング（\0 の頭の右上）: シェルの着せ替え。集中リング（橙赤）と休憩リング（青緑）に、
// 進捗00〜進捗60 の 61 段がある（画像は dev/make-ring.py が作る）。進捗が 1 段進んだ秒に、
// 文字を出さない \![bind] のスクリプトだけを返して切り替える。
//
// 集中中のランダムトークは本体のしくみに任せず、OnSecondChange で時刻を見て出す。
//******************************************************************************

//------------------------------------------------------------------------------
// 起動時の初期化・期限切れの片付け
//------------------------------------------------------------------------------
ゴースト起動前の処理.ポモドーロ : void
{
	if ISVAR('pomo_state') == 0 {
		pomo_state = 0
	}
	if ISVAR('pomo_end') == 0 {
		pomo_end = 0
	}
	if ISVAR('pomo_mid') == 0 {
		pomo_mid = 0
	}
	if ISVAR('pomo_talk_next') == 0 {
		pomo_talk_next = 0
	}
	if ISVAR('pomo_focus_min') == 0 {
		pomo_focus_min = 25
	}
	if ISVAR('pomo_break_min') == 0 {
		pomo_break_min = 5
	}
	if ISVAR('pomo_overrun') == 0 {
		pomo_overrun = 0
	}
	if ISVAR('pomo_type') == 0 {
		pomo_type = 0
	}
	if ISVAR('pomo_oneline') == 0 {
		pomo_oneline = 3
	}
	if ISVAR('pomo_miss') == 0 {
		pomo_miss = 0
	}
	if ISVAR('pomo_aitalk_backup') == 0 {
		pomo_aitalk_backup = 0
	}
	if ISVAR('pomo_interval_override') == 0 {
		pomo_interval_override = 0
	}
	if ISVAR('pomo_custom_focus') == 0 {
		pomo_custom_focus = 25
	}
	if ISVAR('pomo_ring_shown') == 0 {
		pomo_ring_shown = ''
	}

	//ゴーストを閉じている間に終わりの時刻を過ぎていたら、集中は無かったことにする
	//（開き直した直後に、遅れた鐘が鳴らないようにする）
	if pomo_state != 0 {
		if GETSECCOUNT > pomo_end {
			集中リセット
		}
	}
}

//------------------------------------------------------------------------------
// 状態を待機に戻す（本体のランダムトーク間隔も元に戻す）
//------------------------------------------------------------------------------
集中リセット : void
{
	集中間隔復元
	pomo_state = 0
	pomo_end = 0
	pomo_mid = 0
	pomo_talk_next = 0
	pomo_overrun = 0
}

//------------------------------------------------------------------------------
// aitalkinterval を元に戻す（集中中に自分で 0 にしていたときだけ）
//------------------------------------------------------------------------------
集中間隔復元 : void
{
	if pomo_interval_override {
		aitalkinterval = pomo_aitalk_backup
		pomo_interval_override = 0
	}
}

//------------------------------------------------------------------------------
// トーク頻度の設定（メニューから呼ぶ）
// 集中中は aitalkinterval が 0 になっているので、退避した値のほうを書き換える
//------------------------------------------------------------------------------
トーク間隔設定 : void
{
	if pomo_interval_override {
		pomo_aitalk_backup = TOINT(_argv[0])
	}
	else {
		aitalkinterval = TOINT(_argv[0])
	}
}

//------------------------------------------------------------------------------
// 「終わらない」確率（％）。外れが続くほど少しずつ上がる
//------------------------------------------------------------------------------
集中延長確率
{
	_p = 15 + pomo_miss * 5
	if _p > 40 {
		_p = 40
	}
	_p
}

//------------------------------------------------------------------------------
// 集中の開始。_argv[0] = 集中の分数、_argv[1] = 休憩の分数
//------------------------------------------------------------------------------
集中開始
{
	pomo_focus_min = TOINT(_argv[0])
	pomo_break_min = TOINT(_argv[1])
	pomo_state = 1
	pomo_end = GETSECCOUNT + pomo_focus_min * 60
	pomo_mid = 0
	pomo_overrun = 0

	//本体のランダムトークを止める（元の値は退避）
	if pomo_interval_override == 0 {
		pomo_aitalk_backup = aitalkinterval
		aitalkinterval = 0
		pomo_interval_override = 1
	}

	//「黙ってて」にしているときは、集中専用トークも出さない
	if pomo_aitalk_backup > 0 {
		pomo_talk_next = GETSECCOUNT + 300 + RAND(180)
	}
	else {
		pomo_talk_next = 0
	}

	集中開始トーク
}

//------------------------------------------------------------------------------
// 1 秒ごとの判定（あやりりすの AYL_OnSecondChange から呼ばれる）
// 状態を書き換えるのは、トークを返す分岐の中だけ。待機中は何も書き換えない。
//------------------------------------------------------------------------------
OnSecondChange
{
	//待機中で、リングも出ていなければ、何もしない
	if pomo_state == 0 && pomo_ring_shown == '' {
		return
	}

	//しゃべれないとき（最小化・全画面・しゃべり中・選択肢表示中）は、次の秒に回す
	if TOINT(reference[3]) == 0 {
		return
	}
	if SHIORI3FW.CanTalk == 0 {
		return
	}

	//待機に戻ったのにリングが残っていたら消す
	if pomo_state == 0 {
		集中リング更新
		return
	}

	_now = GETSECCOUNT

	if pomo_state == 1 {
		//集中の終わり
		if _now >= pomo_end {
			集中終了
			return
		}
		//集中専用トーク（鐘まで 2 分を切ったら出さない）
		if pomo_talk_next > 0 {
			if _now >= pomo_talk_next {
				if (pomo_end - _now) >= 120 {
					pomo_talk_next = _now + 420 + RAND(300)
					集中中トーク
					return
				}
			}
		}
	}
	elseif pomo_state == 2 {
		//延長中の実況（1 回だけ）
		if pomo_mid > 0 {
			if _now >= pomo_mid {
				pomo_mid = 0
				EVAL("集中延長実況%(pomo_type)")
				return
			}
		}
		//休憩の終わり
		if _now >= pomo_end {
			休憩終了
			return
		}
	}

	//鐘も実況も専用トークも出さない秒に、リングの進みを反映する
	集中リング更新
}

//------------------------------------------------------------------------------
// 進捗リング（シェルの着せ替え）
//------------------------------------------------------------------------------

//---- 今出すべきリングを 'カテゴリ,パーツ' で返す（待機中は ''）
集中リング目標
{
	if pomo_state == 0 {
		''
		return
	}
	//ブロックの中で初めて使ったローカル変数は、ブロックの外で使えない（YAYA の仕様）ので、先に宣言する
	_dur = 0
	_cat = ''
	if pomo_state == 1 {
		_dur = pomo_focus_min * 60
		_cat = '集中リング'
	}
	else {
		_dur = pomo_break_min * 60
		_cat = '休憩リング'
	}
	//区間の始まりからの経過を 60 等分する（0〜60。画像の段数は dev/make-ring.py の STEPS）
	_k = (GETSECCOUNT - (pomo_end - _dur)) * 60 / _dur
	if _k < 0 {
		_k = 0
	}
	if _k > 60 {
		_k = 60
	}
	_n = TOSTR(_k)
	if _k < 10 {
		_n = '0' + _n
	}
	_cat + ',進捗' + _n
}

//---- リングが変わるときだけ、切り替えのスクリプトを返す（変わらなければ ''）
//     先頭の \C は、直前のトークの吹き出しを消さないため。
//     状態（pomo_ring_shown）は、スクリプトを返す分岐の中でだけ書き換える。
集中リング更新
{
	_target = 集中リング目標
	if _target == pomo_ring_shown {
		return
	}
	_s = '\C\0'
	if pomo_ring_shown != '' {
		_s += '\![bind,' + pomo_ring_shown + ',0]'
	}
	if _target != '' {
		_s += '\![bind,' + _target + ',1]'
	}
	pomo_ring_shown = _target
	_s
}

//---- ゴーストを起動して最初のスクリプトの前に、前回のリングを消す
//     （続きの集中があれば、次の秒の OnSecondChange でリングを付け直す）
起動初期化スクリプト
{
	pomo_ring_shown = ''
	'\0\![bind,集中リング,,0]\![bind,休憩リング,,0]'
}

//------------------------------------------------------------------------------
// 集中の終わり → 休憩の始まり
//------------------------------------------------------------------------------
集中終了
{
	pomo_state = 2
	pomo_end = GETSECCOUNT + pomo_break_min * 60
	pomo_talk_next = 0

	if RAND(100) < 集中延長確率 {
		//ぽにゃが終わらない回。本体のランダムトークは休憩が終わるまで止めたままにする
		pomo_miss = 0
		pomo_overrun = 1
		pomo_type = RAND(2)
		pomo_oneline = 3 + RAND(4)
		pomo_mid = GETSECCOUNT + pomo_break_min * 60 * 45 / 100
		EVAL("集中延長鐘%(pomo_type)")
	}
	else {
		pomo_miss++
		pomo_overrun = 0
		pomo_mid = 0
		集中間隔復元
		void SHIORI3FW.ResetAITalkInterval(0)
		集中鐘通常
	}
}

//------------------------------------------------------------------------------
// 休憩の終わり。選択肢を出す前に状態を待機に戻しておく
//------------------------------------------------------------------------------
休憩終了
{
	_overrun = pomo_overrun
	pomo_state = 0
	pomo_mid = 0
	pomo_overrun = 0
	集中間隔復元
	void SHIORI3FW.ResetAITalkInterval(0)

	if _overrun {
		EVAL("集中延長止め%(pomo_type)")
	}
	else {
		休憩終了通常
	}
}

//******************************************************************************
// メニュー
//******************************************************************************

Menu_POMODORO
{
	if pomo_state == 0 {
		集中メニュー選択
	}
	else {
		集中メニュー進行
	}
}

集中メニュー選択
{
	<<"
	ぽ００：どれにするのです？
	＠メニュー：25分集中 / 5分休憩（定番）｜Menu_POMODORO_25
	＠メニュー：50分集中 / 10分休憩｜Menu_POMODORO_50
	＠メニュー：15分集中 / 3分休憩｜Menu_POMODORO_15
	＠メニュー：自分で決める｜Menu_POMODORO_CUSTOM
	＠改行多めメニュー：戻る｜OpenMenu
	">>
}

集中メニュー進行
{
	<<"
	ぽ００【pomo_overrun == 0】：集中タイマー、動いているのです。
	ぽ０６【pomo_overrun != 0】：…
	れ１０【pomo_overrun != 0】：ぽにゃは今、取り込み中や。
	＠メニュー：今どのくらい？｜Menu_POMODORO_STATUS
	＠メニュー：やめる｜Menu_POMODORO_STOP
	＠改行多めメニュー：戻る｜OpenMenu
	">>
}

Menu_POMODORO_25
{
	集中開始(25, 5)
}

Menu_POMODORO_50
{
	集中開始(50, 10)
}

Menu_POMODORO_15
{
	集中開始(15, 3)
}

//---- 休憩のあとの「もう一回」。直前の分数をそのまま使う
Menu_POMODORO_AGAIN
{
	if pomo_state != 0 {
		return
	}
	集中開始(pomo_focus_min, pomo_break_min)
}

Menu_POMODORO_DONE
{
	<<"
	ぽ００：お疲れさまなのです。
	れ１０：ほな、解散や。
	">>
}

//---- 今どのくらい？
Menu_POMODORO_STATUS
{
	_min = (pomo_end - GETSECCOUNT + 59) / 60
	if _min < 1 {
		_min = 1
	}
	<<"
	ぽ００【pomo_state == 1】：集中はあと%(_min)分なのです。
	れ１０【pomo_state == 1】：ぽにゃ、数えとったんか。
	ぽ００【pomo_state == 2 && pomo_overrun == 0】：休憩はあと%(_min)分なのです。
	れ１０【pomo_state == 2 && pomo_overrun == 0】：休憩て、短いなぁ。
	ぽ０６【pomo_state == 2 && pomo_overrun != 0】：…
	れ１０【pomo_state == 2 && pomo_overrun != 0】：休憩はあと%(_min)分や。ぽにゃは知らんけどな。
	">>
}

//---- やめる
Menu_POMODORO_STOP
{
	集中リセット
	void SHIORI3FW.ResetAITalkInterval(0)
	<<"
	ぽ００：やめるのです。また始めるときは言ってほしいのです。
	れ１０：さよか。
	">>
}

//---- 自分で決める（スライダー入力）
Menu_POMODORO_CUSTOM
{
	<<"
	ぽ００：集中する時間を決めるのです。
	＠スクリプト：\![open,sliderinput,OnPomodoroFocusInput,0,25,1,180]
	">>
}

OnPomodoroFocusInput
{
	_v = TOINT(reference[0])
	if _v < 1 || _v > 180 {
		_v = 25
	}
	pomo_custom_focus = _v
	<<"
	ぽ００：%(_v)分なのです。休憩は何分にするのです？
	＠スクリプト：\![open,sliderinput,OnPomodoroBreakInput,0,5,1,60]
	">>
}

OnPomodoroBreakInput
{
	if pomo_state != 0 {
		return
	}
	_b = TOINT(reference[0])
	if _b < 1 || _b > 60 {
		_b = 5
	}
	集中開始(pomo_custom_focus, _b)
}

//---- スライダーを閉じた
OnUserInputCancel
{
	if reference[0] == 'OnPomodoroFocusInput' || reference[0] == 'OnPomodoroBreakInput' {
		集中入力中止
	}
}

集中入力中止
{
	<<"
